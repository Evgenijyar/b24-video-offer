#!/usr/bin/env python3

import argparse
import fcntl
import os
from pathlib import Path
import shlex
import shutil
import subprocess
import sys
import time


BASE = Path("/opt/okk")
REPO = BASE / "repo"
RUNTIME = BASE / "runtime"
DATA = BASE / "data" / "video-cache"
LOCK = Path("/var/lock/okk-deploy.lock")

REPO_URL = "https://github.com/Evgenijyar/OKK.git"
BRANCH = "main"

CONTAINER = "okk-app"
IMAGE_REPOSITORIES = {"okk-app", "deploy-okk"}

HOST_PORT = "8081"
CONTAINER_PORT = "8081"


def command_text(cmd):
    return " ".join(shlex.quote(str(part)) for part in cmd)


def run(cmd, *, check=True, capture=False, quiet=False):
    cmd = [str(part) for part in cmd]

    if not quiet:
        print(f"+ {command_text(cmd)}", flush=True)

    result = subprocess.run(
        cmd,
        text=True,
        stdout=subprocess.PIPE if capture else None,
        stderr=subprocess.PIPE if capture else None,
    )

    if check and result.returncode != 0:
        if capture:
            if result.stdout:
                print(result.stdout, end="", file=sys.stderr)
            if result.stderr:
                print(result.stderr, end="", file=sys.stderr)

        raise subprocess.CalledProcessError(result.returncode, cmd)

    return result


def output(cmd):
    return run(cmd, capture=True).stdout.strip()


def exists_command(name):
    return shutil.which(name) is not None


def docker_container_exists(name):
    result = run(
        ["docker", "container", "inspect", name],
        check=False,
        capture=True,
        quiet=True,
    )
    return result.returncode == 0


def ensure_prerequisites():
    if os.geteuid() != 0:
        raise SystemExit("Run as root.")

    for name in ("docker", "git", "curl", "nginx"):
        if not exists_command(name):
            raise SystemExit(f"Missing required command: {name}")

    run(["docker", "info"], capture=True)

    BASE.mkdir(parents=True, exist_ok=True)


def ensure_swap():
    active_swap = output(["swapon", "--show=NAME", "--noheadings"])

    if active_swap:
        return

    swapfile = Path("/swapfile")

    if not swapfile.exists() or swapfile.stat().st_size < 2 * 1024**3:
        run(["swapoff", str(swapfile)], check=False)
        swapfile.unlink(missing_ok=True)

        result = run(
            ["fallocate", "-l", "2G", str(swapfile)],
            check=False,
        )

        if result.returncode != 0:
            run(
                [
                    "dd",
                    "if=/dev/zero",
                    f"of={swapfile}",
                    "bs=1M",
                    "count=2048",
                    "status=progress",
                ]
            )

    run(["chmod", "600", str(swapfile)])
    run(["mkswap", str(swapfile)])
    run(["swapon", str(swapfile)])

    fstab = Path("/etc/fstab")
    content = fstab.read_text(encoding="utf-8") if fstab.exists() else ""

    swap_already_configured = any(
        line.split()[:1] == [str(swapfile)]
        for line in content.splitlines()
        if line.strip() and not line.lstrip().startswith("#")
    )

    if not swap_already_configured:
        with fstab.open("a", encoding="utf-8") as handle:
            handle.write(f"{swapfile} none swap sw 0 0\n")

    Path("/etc/sysctl.d/99-okk-swap.conf").write_text(
        "vm.swappiness=10\n",
        encoding="utf-8",
    )

    run(["sysctl", "-p", "/etc/sysctl.d/99-okk-swap.conf"])


def remove_matching_images():
    result = run(
        ["docker", "image", "ls", "--format", "{{.Repository}}:{{.Tag}}"],
        capture=True,
        check=False,
        quiet=True,
    )

    for reference in result.stdout.splitlines():
        repository = reference.rsplit(":", 1)[0]

        if repository in IMAGE_REPOSITORIES:
            run(
                ["docker", "image", "rm", "-f", reference],
                check=False,
            )


def reset_old_deployment():
    compose_file = REPO / "deploy" / "docker-compose.yml"

    if compose_file.exists():
        run(
            [
                "docker",
                "compose",
                "-f",
                compose_file,
                "down",
                "--remove-orphans",
                "--volumes",
            ],
            check=False,
        )

    if docker_container_exists(CONTAINER):
        run(["docker", "rm", "-f", CONTAINER], check=False)

    for volume_name in (
        "deploy_okk_video_cache",
        "okk-video-cache",
    ):
        run(
            ["docker", "volume", "rm", "-f", volume_name],
            check=False,
        )

    run(
        ["docker", "network", "rm", "deploy_default"],
        check=False,
    )

    remove_matching_images()

    for path in (
        REPO,
        RUNTIME,
        BASE / "data",
    ):
        if path.exists():
            shutil.rmtree(path)

    run(["docker", "image", "prune", "-f"], check=False)


def update_repository():
    if (REPO / ".git").is_dir():
        run(
            [
                "git",
                "-C",
                REPO,
                "remote",
                "set-url",
                "origin",
                REPO_URL,
            ]
        )

        run(
            [
                "git",
                "-C",
                REPO,
                "fetch",
                "--prune",
                "--depth",
                "1",
                "origin",
                BRANCH,
            ]
        )

        run(
            [
                "git",
                "-C",
                REPO,
                "reset",
                "--hard",
                "FETCH_HEAD",
            ]
        )

        run(
            [
                "git",
                "-C",
                REPO,
                "clean",
                "-fdx",
            ]
        )

    else:
        if REPO.exists():
            shutil.rmtree(REPO)
